================================= // 🚀 ระบบประมวลผลและแสดงผล (ไม่ต้องแก้ไขส่วนนี้) // ========================================================================= const { createApp, ref, computed } = Vue; createApp({ setup() { const availableYears = [2570, 2571, 2572]; const selectedYear = ref(2570); const currentQuota = computed(() => { const q = quotasData[selectedYear.value] || { vacationCarried: 0, vacationBase: 10, personalMax: 45, sickMax: 60 }; return { ...q, vacationTotal: (q.vacationCarried || 0) + (q.vacationBase || 10) }; }); const currentYearLeaves = computed(() => { return leavesData.filter(item => item.fiscalYear === selectedYear.value); }); const vacationUsed = computed(() => { return currentYearLeaves.value .filter(l => l.type === 'ลาพักผ่อน') .reduce((sum, l) => sum + l.days, 0); }); const vacationRemaining = computed(() => { return Math.max(0, currentQuota.value.vacationTotal - vacationUsed.value); }); const personalUsed = computed(() => { return currentYearLeaves.value .filter(l => l.type === 'ลากิจส่วนตัว') .reduce((sum, l) => sum + l.days, 0); }); const personalRemaining = computed(() => { return Math.max(0, currentQuota.value.personalMax - personalUsed.value); }); const sickUsed = computed(() => { return currentYearLeaves.value .filter(l => l.type === 'ลาป่วย') .reduce((sum, l) => sum + l.days, 0); }); const sickRemaining = computed(() => { return Math.max(0, currentQuota.value.sickMax - sickUsed.value); }); const otherUsed = computed(() => { return currentYearLeaves.value .filter(l => !['ลาพักผ่อน', 'ลากิจส่วนตัว', 'ลาป่วย'].includes(l.type)) .reduce((sum, l) => sum + l.days, 0); }); const totalDaysUsed = computed(() => { return currentYearLeaves.value.reduce((sum, l) => sum + l.days, 0); }); const getTypeBadge = (type) => { if (type === 'ลาพักผ่อน') return 'bg-emerald-500'; if (type === 'ลากิจส่วนตัว') return 'bg-blue-500'; if (type === 'ลาป่วย') return 'bg-rose-500'; return 'bg-purple-500'; }; const getTypeTag = (type) => { if (type === 'ลาพักผ่อน') return 'bg-emerald-50 text-emerald-700'; if (type === 'ลากิจส่วนตัว') return 'bg-blue-50 text-blue-700'; if (type === 'ลาป่วย') return 'bg-rose-50 text-rose-700'; return 'bg-purple-50 text-purple-700'; }; return { availableYears, selectedYear, currentQuota, currentYearLeaves, vacationUsed, vacationRemaining, personalUsed, personalRemaining, sickUsed, sickRemaining, otherUsed, totalDaysUsed, getTypeBadge, getTypeTag }; } }).mount('#app');
